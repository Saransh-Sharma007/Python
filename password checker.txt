s = input().strip()
if len(s) < 6 or len(s) > 12:
    print("INVALID")
else:
    has_digit = False
    has_upper = False
    has_lower = False

    for ch in s:
        if ch.isdigit():
            has_digit = True
        elif ch.isupper():
            has_upper = True
        elif ch.islower():
            has_lower = True

    if has_digit and has_upper and has_lower:
        print("VALID")
    else:
        print("INVALID")
