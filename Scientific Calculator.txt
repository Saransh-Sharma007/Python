class ScientificCalculator:
    def __init__(self, x=0, y=0, z=0):
        self.x = x
        self.y = y
        self.z = z

    def __str__(self):
        return f"({self.x}, {self.y}, {self.z})"

    def __add__(self, other):
        return ScientificCalculator(self.x + other.x, self.y + other.y, self.z + other.z)

    def __mul__(self, other):
        # Dot product
        return self.x * other.x + self.y * other.y + self.z * other.z

    def __xor__(self, other):
        # Cross product
        cx = self.y * other.z - self.z * other.y
        cy = self.z * other.x - self.x * other.z
        cz = self.x * other.y - self.y * other.x
        return ScientificCalculator(cx, cy, cz)


if __name__ == "__main__":
    print("Enter components of first vector (x y z): ")
    x1, y1, z1 = map(float, input().split())
    print("Enter components of second vector (x y z): ")
    x2, y2, z2 = map(float, input().split())

    v1 = ScientificCalculator(x1, y1, z1)
    v2 = ScientificCalculator(x2, y2, z2)

    print("\nChoose the operation to perform:")
    print("1. Addition (+)")
    print("2. Dot Product (*)")
    print("3. Cross Product (^)")
    
    choice = input("\nEnter your choice (1/2/3 or +/*/^): ").strip()

    if choice in ['1', '+']:
        print("\nAddition Result:", v1 + v2)
    elif choice in ['2', '*']:
        print("\nDot Product Result:", v1 * v2)
    elif choice in ['3', '^']:
        print("\nCross Product Result:", v1 ^ v2)
    else:
        print("\nInvalid choice. Please select +, *, or ^.")
